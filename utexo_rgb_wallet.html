import { WalletManagerRgb } from '@utexo/wdk-wallet-rgb';

// 1. Initialize WDK RGB Wallet Manager
const manager = new WalletManagerRgb({
  network: 'testnet',
  indexerUrl: 'https://electrs.rgb.info',
  transportEndpoint: 'rpca://storm.rgb.info:50001',
  dataDir: '~/.wdk/rgb-wallet/testnet'
});

// 2. Create Taproot account from BIP-39 mnemonic
const account = await manager.createAccount({
  mnemonic: 'twelve word mnemonic phrase...',
  derivationPath: "m/86'/1'/0'/0/0"
});

// 3. Generate blinded UTXO invoice
const invoice = await account.createBlindedInvoice({
  assetId: 'rgb:nia:3b9f8e41a27c09d854e12e7f83a54b92c431ef82a',
  amount: 50
});

// 4. Execute 3-stage transfer pipeline
const prep = await account.sendBegin({ recipient: invoice.blindedUtxo, amount: 25 });
const signedPsbt = await account.signPsbt(prep.psbt);
const result = await account.sendEnd({ signedPsbt, consignment: prep.consignment });
