Official Tether WDK (@tetherto/wdk) Usage

import { WDK } from "@tetherto/wdk";
import { UtexoWalletRGB } from "@utexo/wdk-wallet-rgb";

// 1. Initialize WDK with Utexo RGB Module
const wdk = new WDK({
  network: "testnet",
  indexerUrl: "https://rgb.indexer.utexo.io"
});

wdk.registerWallet("rgb", new UtexoWalletRGB({
  seedPhrase: "twelve secret recovery words here ...",
  derivationPath: "m/86'/1'/0'/0/0" // BIP-86 Taproot
}));

// 2. Fetch UTXOs with RGB allocations
const wallet = wdk.getWallet("rgb");
const utxos = await wallet.getUtxos();
console.log("RGB UTXOs:", utxos);
